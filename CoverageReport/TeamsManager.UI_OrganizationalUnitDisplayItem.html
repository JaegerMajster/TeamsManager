 744        /// <param name="units">Lista jednostek do uporządkowania</param>
 745        private void BuildHierarchicalStructure(List<OrganizationalUnit> units)
 746        {
 747            // Utwórz słownik dla szybkiego wyszukiwania
 748            var unitDict = units.ToDictionary(u => u.Id, u => u);

 752                                .OrderBy(u => u.SortOrder)
